\hline
\textbf{Từ viết tắt} & \textbf{Diễn giải} \\
\hline
\endhead
ACF & Autocorrelation Function -- hàm tự tương quan \\ \hline
BKT & Bài kiểm tra (BKT1: Tuần 4, BKT2: Tuần 6, BKT3: Tuần 8) \\ \hline
CV & Cross-Validation -- kiểm định chéo \\ \hline
EDA & Exploratory Data Analysis -- phân tích khám phá dữ liệu \\ \hline
GAP & \code{Global_active_power} -- công suất tác dụng toàn hộ (kW) \\ \hline
IQR & Interquartile Range -- khoảng tứ phân vị ($Q_3 - Q_1$) \\ \hline
kWh & Kilowatt giờ -- đơn vị điện năng \\ \hline
MAE & Mean Absolute Error -- sai số tuyệt đối trung bình \\ \hline
MAPE & Mean Absolute Percentage Error -- sai số phần trăm tuyệt đối trung bình \\ \hline
MASE & Mean Absolute Scaled Error -- sai số tuyệt đối trung bình có chuẩn hóa \\ \hline
MCAR & Missing Completely At Random -- thiếu hoàn toàn ngẫu nhiên \\ \hline
NaN & Not a Number -- giá trị khuyết thiếu \\ \hline
RMSE & Root Mean Squared Error -- căn bậc hai sai số bình phương trung bình \\ \hline
SNaive & Seasonal Naive -- mô hình cơ sở lặp lại giá trị cùng mùa \\ \hline
TC & Tiêu chí chấm của học phần \\ \hline
UCI & University of California, Irvine (Kho dữ liệu Học máy UCI) \\ \hline
\end{longtable}
